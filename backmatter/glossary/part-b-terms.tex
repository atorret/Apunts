\phantomsection\subsection*{Part B. Terms}\addcontentsline{toc}{subsection}{Part B. Terms}

Alphabetical. Each entry gives a one-line definition and the place where the term is defined; ``see'' entries are synonyms.

\medskip
\begingroup\small\setlength{\parskip}{2pt}
\glossentry{Alternating map.}{A multilinear map that changes sign when two arguments are exchanged; it vanishes if two arguments coincide.}{sec:altforms}
\glossentry{Argument principle.}{$\frac1{2\pi i}\oint f'/f\,\dd z$ counts zeros minus poles inside the contour.}{sec:argument}
\glossentry{Atlas.}{A family of charts covering a manifold with smooth transition maps.}{sec:manifolds}
\glossentry{Bianchi identity.}{First: the cyclic sum of $R(X,Y)Z$ vanishes for a torsion-free connection; second: the cyclic sum of $\nabla\mathbb R$ vanishes.}{sec:curv}
\glossentry{Blade.}{A multivector that is a wedge of vectors, $a_1\wedge\cdots\wedge a_k$; it represents an oriented subspace.}{sec:blades}
\glossentry{Boundary operator.}{The map $c\mapsto\partial c$ on chains; Stokes says it is adjoint to $\dd$.}{sec:adjoint}
\glossentry{Cartan formula.}{$\Lie_X=\dd\,\iota_X+\iota_X\,\dd$ on differential forms.}{sec:cartan}
\glossentry{Cauchy--Pompeiu formula.}{Cauchy's integral formula for a smooth, not necessarily holomorphic, function, with an area term in $\partial_{\bar z}f$.}{sec:pompeiu}
\glossentry{Cauchy--Riemann equations.}{$\partial_{\bar z}f=0$; equivalently $\dd f$ is complex-linear, or $\vd\mathbf u=0$.}{sec:complexforms}
\glossentry{Cauchy theorem.}{The integral of a holomorphic $f\,\dd z$ over the boundary of a region is zero: Stokes for a closed form.}{sec:cauchy}
\glossentry{Chain.}{A formal sum of oriented $k$-dimensional pieces, over which $k$-forms are integrated.}{sec:adjoint}
\glossentry{Chart.}{A homeomorphism from an open set of $M$ onto an open set of $\R^m$; it assigns coordinates.}{sec:manifolds}
\glossentry{Christoffel symbols.}{The coefficients $\Gamma^k{}_{ij}$ of a connection in a chart, $\nabla_{\partial_i}\partial_j=\Gamma^k{}_{ij}\partial_k$; not a tensor.}{sec:conn}
\glossentry{Clifford algebra.}{See \emph{geometric algebra}.}{}
\glossentry{Closed form.}{A form with $\dd\omega=0$.}{sec:expl-closed}
\glossentry{Codifferential.}{The operator $\codiff=\pm\hodge\dd\hodge$, lowering degree by one; in geometric algebra $-\vd\lc$.}{sec:ga-calculus}
\glossentry{Components.}{The numbers that express a vector, covector or tensor in a basis.}{sec:vs}
\glossentry{Connection (affine).}{A rule $\nabla_XY$ for differentiating vector fields, $C^\infty$-linear in $X$ and Leibniz in $Y$.}{sec:conn}
\glossentry{Constitutive relation.}{The metric-dependent link between field forms of a medium, \eg $D=\varepsilon_0\hodge E$.}{sec:phys-em}
\glossentry{Contraction.}{Feeding a vector into a covector slot: $\omega(v)$, a trace of a tensor, $\iota_v$, or $a\lc A$.}{sec:contr4}
\glossentry{Contravariant.}{Transforming like vector components (with the inverse of the change-of-basis matrix); upper indices.}{sec:tensortypes}
\glossentry{Coordinate basis.}{The vectors $\partial_i$ and covectors $\dd x^i$ defined by a chart.}{sec:tangentvectors}
\glossentry{Cotangent space.}{The dual $T_p^*M$ of the tangent space; its elements are covectors at $p$.}{sec:tangent}
\glossentry{Covariant.}{Transforming like covector components (with the change-of-basis matrix itself); lower indices.}{sec:tensortypes}
\glossentry{Covariant derivative.}{See \emph{connection}.}{}
\glossentry{Covector.}{An element of the dual space $V^*$, \ie a linear map $V\to\R$; also called a $1$-form or linear form.}{sec:dualspace}
\glossentry{Cross product.}{In $\R^3$ only, $a\times b=\dual{(a\wedge b)}$; needs a metric and an orientation.}{sec:duality}
\glossentry{Curvature.}{See \emph{Riemann tensor}.}{}
\glossentry{de Rham cohomology.}{Closed forms modulo exact forms; it detects holes, \eg $H^1_{\mathrm{dR}}(\R^2\setminus\{0\})\cong\R$.}{sec:adjoint}
\glossentry{Derivation.}{A linear map on functions satisfying the Leibniz rule; tangent vectors are derivations at a point.}{sec:tangentvectors}
\glossentry{Diffeomorphism.}{A smooth bijection with smooth inverse.}{sec:pushpull}
\glossentry{Differential form.}{A smooth field of alternating multilinear maps on tangent vectors.}{sec:dform}
\glossentry{Divergence.}{The relative rate of change of volume under a flow, $\Lie_X\mu=(\operatorname{div}_\mu X)\mu$.}{sec:divmu}
\glossentry{Divergence theorem.}{See \emph{Gauss theorem}.}{}
\glossentry{Dual (geometric algebra).}{$\dual A=A\pss^{-1}$; it maps a blade to its orthogonal complement.}{sec:duality}
\glossentry{Dual basis.}{The covectors $e^i$ with $e^i(e_j)=\delta^i_j$.}{sec:dualspace}
\glossentry{Dual space.}{The space $V^*$ of linear maps $V\to\R$.}{sec:dualspace}
\glossentry{Einstein tensor.}{$G_{\mu\nu}=\Ric_{\mu\nu}-\frac12s\,g_{\mu\nu}$, divergence-free by the contracted Bianchi identity.}{sec:bridge}
\glossentry{Euler--Lagrange equations.}{The equations of a critical point of an action; for $\frac12g(\dot x,\dot x)$ they are the geodesic equations.}{sec:lc}
\glossentry{Exact form.}{A form $\omega=\dd\eta$.}{sec:expl-closed}
\glossentry{Exterior algebra.}{$\bigwedge V$, the algebra of the wedge product; the quotient of $T(V)$ by $v\otimes v$.}{sec:ext}
\glossentry{Exterior derivative.}{The operator $\dd:\Omega^k\to\Omega^{k+1}$ with $\dd f$ the differential, the graded Leibniz rule and $\dd^2=0$.}{sec:dform}
\glossentry{Exterior product.}{See \emph{wedge product}.}{}
\glossentry{Flow.}{The family of maps $\phi_t$ obtained by solving $\dot\gamma=X_\gamma$.}{sec:lie}
\glossentry{Flux.}{The integral of $\iota_X\mu$ over a hypersurface: how much of $X$ crosses it.}{sec:flux}
\glossentry{Gauss theorem.}{$\int_M\operatorname{div}X\,\dd V=\int_{\partial M}\langle X,N\rangle\,\dd A$: Stokes for $\iota_X\vol_g$.}{sec:gauss}
\glossentry{Geodesic.}{A curve with $\nabla_{\dot\gamma}\dot\gamma=0$: it parallel-transports its own velocity.}{sec:conn}
\glossentry{Geodesic deviation.}{The equation $D_t^2\xi=R(\dot\gamma,\xi)\dot\gamma$ for the separation of nearby geodesics.}{sec:deviation}
\glossentry{Geometric algebra.}{$\Cl(V,Q)=T(V)/\langle v\otimes v-Q(v)\rangle$, the algebra of the geometric product.}{sec:clifford}
\glossentry{Geometric product.}{The product of $\Cl(V,Q)$; for vectors $ab=a\cdot b+a\wedge b$.}{sec:clifford}
\glossentry{Grade.}{The degree $k$ of a $k$-vector; $\gr{A}{k}$ is the grade-$k$ part.}{sec:ops}
\glossentry{Grade involution.}{The automorphism $\hat A$ with $\hat v=-v$; $(-1)^k$ on $k$-vectors.}{sec:ops}
\glossentry{Green theorem.}{$\oint_{\partial D}P\,\dd x+Q\,\dd y=\iint_D(Q_x-P_y)\,\dd A$: Stokes in the plane.}{sec:green}
\glossentry{Hodge star.}{The map $\hodge:\bigwedge^k\to\bigwedge^{n-k}$ with $\alpha\wedge\hodge\beta=\langle\alpha,\beta\rangle\vol$.}{sec:duality}
\glossentry{Holomorphic.}{Complex differentiable; $\partial_{\bar z}f=0$; equivalently $f\,\dd z$ is closed.}{sec:complexforms}
\glossentry{Induced metric.}{The pull-back of a metric to a submanifold.}{sec:pushpull}
\glossentry{Induced orientation.}{The orientation of $\partial M$ for which the outward normal followed by a positive basis of $T\partial M$ is positive.}{sec:stokes}
\glossentry{Inertia metric.}{The kinetic metric $g$ with $T=\frac12g(v,v)$ (mass matrix); it converts velocities into momenta.}{sec:phys-work}
\glossentry{Interior product.}{$\iota_v\alpha$: insert $v$ in the first slot of a form.}{sec:interior}
\glossentry{Kelvin--Stokes theorem.}{$\oint_{\partial S}F\cdot\dd r=\iint_S\operatorname{curl}F\cdot n\,\dd A$.}{sec:classical}
\glossentry{Killing field.}{A vector field with $\Lie_Xg=0$: an infinitesimal isometry.}{sec:lie}
\glossentry{Koszul formula.}{The formula expressing $g(\nabla_XY,Z)$ through $g$ and brackets; it proves uniqueness of Levi-Civita.}{sec:lc}
\glossentry{$k$-form.}{An alternating $k$-linear map on vectors (algebraic); a field of them (differential).}{sec:altforms}
\glossentry{Laurent series.}{The expansion $\sum_nc_n(z-a)^n$ near an isolated singularity; $c_{-1}$ is the residue.}{sec:residue}
\glossentry{Left contraction.}{$A_r\lc B_s=\gr{A_rB_s}{s-r}$; for a vector it is the metric twin of $\iota_v$.}{sec:ops}
\glossentry{Levi-Civita connection.}{The unique torsion-free connection with $\nabla g=0$.}{sec:lc}
\glossentry{Lie bracket.}{$[X,Y]f=X(Yf)-Y(Xf)$; equals $\Lie_XY$.}{sec:lie}
\glossentry{Lie derivative.}{$\Lie_X\tau=\frac{\dd}{\dd\varepsilon}\big|_0\phi_\varepsilon\pb\tau$: the change of $\tau$ along the flow of $X$.}{sec:lie}
\glossentry{Line integral.}{$\int_\gamma\omega=\int\omega(\dot\gamma)\,\dd t$.}{sec:int1def}
\glossentry{Manifold.}{A space covered by charts with smooth transition maps.}{sec:manifolds}
\glossentry{Manifold with boundary.}{As a manifold, but with charts valued in a half-space; the boundary points form $\partial M$.}{sec:stokes}
\glossentry{Meet.}{The regressive product $A\meet B$, representing the intersection of subspaces.}{sec:duality}
\glossentry{Metric.}{A symmetric non-degenerate bilinear form $g$ on $V$ (or a smooth field of them).}{sec:metric}
\glossentry{Monogenic function.}{A multivector field $F$ with $\vd F=0$; in the plane, a holomorphic function.}{sec:synth4}
\glossentry{Multilinear map.}{A map linear in each argument separately.}{sec:multilinear}
\glossentry{Musical isomorphisms.}{$\flat:v\mapsto g(v,\cdot)$ and its inverse $\sharp$: lowering and raising indices.}{sec:metric}
\glossentry{Noether theorem.}{A symmetry $\xi$ of the Lagrangian gives the conserved quantity $p(\xi)$.}{sec:phys-energy}
\glossentry{Normal coordinates.}{Coordinates around $p$ in which $g_{ij}(p)$ is standard and $\Gamma(p)=0$.}{sec:curv}
\glossentry{Orientation.}{An atlas with positive Jacobians, or a nowhere-vanishing top form up to positive multiples.}{sec:intk}
\glossentry{Parallel transport.}{Moving a vector along a curve with $\nabla_{\dot\gamma}Y=0$.}{sec:conn}
\glossentry{Partition of unity.}{Smooth functions $\rho_\alpha\ge0$, supported in the sets of a cover, locally finite, with $\sum\rho_\alpha=1$.}{sec:intk}
\glossentry{Pin group.}{The group of versors $U$ with $U\rev U=\pm1$.}{sec:rotors}
\glossentry{Poincar\'e lemma.}{On a contractible domain every closed form of degree $\ge1$ is exact.}{sec:dform}
\glossentry{Polarization.}{Recovering $g$ from $Q$: $g(u,v)=\frac12\bigl(Q(u+v)-Q(u)-Q(v)\bigr)$.}{sec:quadform}
\glossentry{Pseudoscalar.}{The unit top-grade element $\pss$ of $\GA(V)$.}{sec:duality}
\glossentry{Pull-back.}{$(\phi\pb\omega)(v,\dots)=\omega(\dd\phi\,v,\dots)$: precompose a form with a map.}{sec:pushpull}
\glossentry{Pure tensor.}{A tensor of the form $v\otimes w$.}{sec:tensorproduct}
\glossentry{Push-forward.}{The differential $T_p\Phi$ of a map, acting on tangent vectors.}{sec:pushpull}
\glossentry{Quadratic form.}{$Q(v)=g(v,v)$.}{sec:quadform}
\glossentry{Reciprocal frame.}{The vectors $e^i_\sharp$ with $e^i_\sharp\cdot e_j=\delta^i_j$; the metric image of the dual basis.}{rem:reciprocal}
\glossentry{Reparametrisation.}{Composing a curve with a diffeomorphism of the parameter interval.}{sec:linearity}
\glossentry{Residue.}{$\Res_af=\frac1{2\pi i}\oint f\,\dd z$ around a small circle; the coefficient $c_{-1}$.}{sec:residue}
\glossentry{Reverse.}{The anti-automorphism $\rev A$ with $\rev v=v$; $(-1)^{k(k-1)/2}$ on $k$-vectors.}{sec:ops}
\glossentry{Ricci tensor.}{The contraction $R_{jk}=R^i{}_{ijk}$ of the Riemann tensor.}{sec:curv}
\glossentry{Riemann tensor.}{$R(X,Y)Z=\nabla_X\nabla_YZ-\nabla_Y\nabla_XZ-\nabla_{[X,Y]}Z$.}{sec:curv}
\glossentry{Rotor.}{An even versor $\rotor$ with $\rotor\rev\rotor=1$; it rotates by $x\mapsto\rotor x\rev\rotor$.}{sec:rotors}
\glossentry{Scalar curvature.}{$s=g^{jk}\Ric_{jk}$.}{sec:curv}
\glossentry{Signature.}{The numbers $(p,q,r)$ of positive, negative and zero directions of $Q$.}{sec:quadform}
\glossentry{Spin group.}{The even part of the Pin group; it double-covers the rotations.}{sec:rotors}
\glossentry{Stokes theorem.}{$\int_M\dd\omega=\int_{\partial M}j\pb\omega$.}{sec:stokes}
\glossentry{Tangent space.}{The space $T_pM$ of derivations (velocities) at $p$.}{sec:tangent}
\glossentry{Tensor algebra.}{$T(V)=\bigoplus_kV^{\otimes k}$ with the tensor product.}{sec:tensoralgebra}
\glossentry{Tensor field.}{A smooth assignment of a tensor at each point of $M$.}{sec:tensorfields}
\glossentry{Tensor of type $(\ell,k)$.}{A multilinear map of $k$ vectors and $\ell$ covectors.}{sec:tensortypes}
\glossentry{Tensor product.}{The universal bilinear product $V\times W\to V\otimes W$.}{sec:tensorproduct}
\glossentry{Tidal operator.}{The map $\xi\mapsto R(\dot\gamma,\xi)\dot\gamma$ of geodesic deviation.}{sec:deviation}
\glossentry{Torsion.}{$\Tor(X,Y)=\nabla_XY-\nabla_YX-[X,Y]$.}{sec:lc}
\glossentry{Transition map.}{The change of coordinates $\varphi_\beta\circ\varphi_\alpha^{-1}$ between overlapping charts.}{sec:manifolds}
\glossentry{Twisted form.}{A form that also needs an outer orientation (\eg charge density, $D$, $H$).}{sec:phys-em}
\glossentry{Vector derivative.}{$\vd=e^i_\sharp\partial_i$; $\vd F=\vd\lc F+\vd\wedge F$.}{sec:ga-calculus}
\glossentry{Vector field.}{A smooth assignment of a tangent vector at each point.}{sec:fields}
\glossentry{Versor.}{A geometric product of invertible vectors; it acts by a composition of reflections.}{sec:rotors}
\glossentry{Volume form.}{A nowhere-vanishing top-degree form $\mu$; for a metric, $\vol_g=\sqrt{|g|}\,\dd x^1\wedge\cdots\wedge\dd x^n$.}{sec:divmu}
\glossentry{Wedge product.}{The antisymmetrised product of forms or multivectors; $\alpha\wedge\beta=(-1)^{kl}\beta\wedge\alpha$.}{sec:altforms}
\glossentry{Winding number.}{$\frac1{2\pi}\oint\dd\theta$: how many times a closed curve goes around a point.}{sec:expl-closed}
\endgroup
