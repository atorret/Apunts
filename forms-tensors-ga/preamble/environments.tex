% =====================================================================
%  Theorem-like environments (plain amsthm look, as in the reference)
%  One counter per section, shared by theorems, examples, remarks, ...
% =====================================================================
\newcounter{thm}[section]
\renewcommand{\thethm}{\thesection.\arabic{thm}}
\numberwithin{equation}{section}
\theoremstyle{definition}
\newtheorem{definition}[thm]{Definition}
\theoremstyle{plain}
\newtheorem{theorem}[thm]{Theorem}
\newtheorem{proposition}[thm]{Proposition}
\newtheorem{lemma}[thm]{Lemma}
\newtheorem{corollary}[thm]{Corollary}

% =====================================================================
%  Colour callout boxes (the visual vocabulary of this document)
%  Shape follows the reference PDF: dark title bar, pale body, rounded.
% =====================================================================
\tcbset{calloutbase/.style={enhanced,breakable,arc=2pt,boxrule=0.6pt,
   left=6pt,right=6pt,top=4pt,bottom=4pt,coltitle=white,
   fonttitle=\bfseries\small,toptitle=1.5pt,bottomtitle=1.5pt,
   before skip=8pt,after skip=8pt,
   segmentation style={solid},
   parbox=false}}

% numbered, shared counter: Example 3.4 (optional name).
% The counter is stepped outside the box. With use counter*, tcolorbox stepped it
% only for the title, so \label inside the box captured the enclosing subsection
% number (docs/AUTHOR_NOTES.md #14).
\newtcolorbox{examplebox}[1][]{calloutbase,
   colback=examplebg,colframe=examplefr,
   title={\hypersetup{linkcolor=white}Example~\thethm\if\relax\detokenize{#1}\relax\else\ \textnormal{---}\ #1\fi}}
\newenvironment{example}[1][]{\refstepcounter{thm}\begin{examplebox}[#1]}{\end{examplebox}}
\newtcolorbox{remarkbox}[1][]{calloutbase,
   colback=remarkbg,colframe=remarkfr,
   title={\hypersetup{linkcolor=white}Remark~\thethm\if\relax\detokenize{#1}\relax\else\ \textnormal{---}\ #1\fi}}
\newenvironment{remark}[1][]{\refstepcounter{thm}\begin{remarkbox}[#1]}{\end{remarkbox}}
\newtcolorbox{specialbox}[1][]{calloutbase,
   colback=specialbg,colframe=specialfr,
   title={\hypersetup{linkcolor=white}Special case~\thethm\if\relax\detokenize{#1}\relax\else\ \textnormal{---}\ #1\fi}}
\newenvironment{specialcase}[1][]{\refstepcounter{thm}\begin{specialbox}[#1]}{\end{specialbox}}
