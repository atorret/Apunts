\subsection{Suggested path through the references}
\begin{itemize}
\item \emph{eigenchris, ``Tensor Calculus'' (YouTube playlist)}: visual explanations of forward/backward transformations, covectors as stacks, tensor products, metric tensor, covariant derivative and Christoffel symbols.
\item J.~M. Lee, \emph{Introduction to Smooth Manifolds} and \emph{Riemannian Manifolds} \cite{Lee}: formal proofs of the material of Parts~\ref{part:manifolds}--\ref{part:geometry}.
\item S.~Carroll, \emph{Spacetime and Geometry} \cite{Carroll}: physics-oriented derivations; ch.~1--3.
\item Doran--Lasenby \cite{DL} and Hestenes--Sobczyk \cite{HS} for the geometric algebra of Part~\ref{part:algebra} and Section~\ref{sec:ga-calculus}.
\item M.~Spivak, \emph{Calculus on Manifolds} \cite{Spivak}, for the cleanest proof of Stokes' theorem.
\end{itemize}
